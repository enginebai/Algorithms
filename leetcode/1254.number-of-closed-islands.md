# [1254. Number of Closed Islands](https://leetcode.com/problems/number-of-closed-islands/)

## Traversal From Islands
We can traverse all the islands, and check if current island is closed. We can use DFS or BFS to traverse the grid. **Please note that we have to traverse all the cells of the island to check if it's closed even if we found a cell that is not closed**. We don't early return, we have to traverse all the cells of the island to mark them as visited.

```kotlin
fun closedIsland(grid: Array<IntArray>): Int {
    val m = grid.size
    val n = grid[0].size
    val visited = Array(m) { BooleanArray(n) }
    var count = 0
    for (i in 0 until m) {
        for (j in 0 until n) {
            if (grid[i][j] == 0 && !visited[i][j]) {
                if (dfs(grid, i, j, visited)) count++
                // if (bfs(grid, i, j, visited)) count++
            }
        }
    }
    return count
}

private fun dfs(grid: Array<IntArray>, x: Int, y: Int, visited: Array<BooleanArray>): Boolean {
    val m = grid.size
    val n = grid[0].size
    if (x !in 0 until m || y !in 0 until n) return false
    if (visited[x][y]) return true
    if (grid[x][y] == 1) return true

    visited[x][y] = true
    var result = true
    for (d in directions) {
        val newX = x + d[0]
        val newY = y + d[1]
        // result = result & dfs(grid, x + d[0], y + d[1]) will not work, we have to traverse all the cells and then return the result.
        // but we use `&&` to short-circuit the evaluation, it won't traverse all the cells of the same island.
        result = dfs(grid, newX, newY, visited) && result
    }
    return result
}

private fun bfs(grid: Array<IntArray>, sourceX: Int, sourceY: Int, visited: Array<BooleanArray>): Boolean {
    val m = grid.size
    val n = grid[0].size
    val queue = ArrayDeque<Pair<Int, Int>>()
    queue.addLast(sourceX to sourceY)
    visited[sourceX][sourceY] = true
    var result = true
    while (queue.isNotEmpty()) {
        val (x, y) = queue.removeFirst()
        for (d in directions) {
            val newX = x + d[0]
            val newY = y + d[1]
            // We don't return false when we find a cell that is not closed, we have to traverse all the cells of the island to mark them as visited.
            if (newX !in 0 until m || newY !in 0 until n) result = false
            else if (visited[newX][newY]) continue
            else if (grid[newX][newY] == 1) continue
            else {
                queue.addLast(newX to newY)
                visited[newX][newY] = true
            }
        }
    }
    return result
}
```

## Traversal From Boundary
To find the closed island, we can find open islands first, then find the closed islands. We can start searching from the edges of the grid, the islands which are connected to the edges are non-closed islands. We can mark them as invalid during traversal. Then we can find the closed islands by traversing the grid again.

* We traversal (either DFS or BFS) from the 4 edges to search **non-closed** islands and mark the region as invalid during traversal.
* Then we traversal the graph again to find `0`, that would be closed islands.

```kotlin
private val invalid = 2

fun closedIsland(grid: Array<IntArray>): Int {
    val m = grid.size
    val n = grid[0].size
    val visited = Array(m) { BooleanArray(n) }

    for (row in 0 until m) {
        dfs(grid, row, 0, visited)
        dfs(grid, row, n - 1, visited)
    }

    for (col in 0 until n) {
        dfs(grid, 0, col, visited)
        dfs(grid, m - 1, col, visited)
    }

    var count = 0
    for (row in 0 until m) {
        for (col in 0 until n) {
            if (grid[row][col] == 0 && !visited[row][col]) {
                dfs(grid, row, col, visited)
                count++
            }
        }
    }
    return count
}

private fun dfs(grid: Array<IntArray>, x: Int, y: Int, visited: Array<BooleanArray>) {
    val m = grid.size
    val n = grid[0].size
    if (x !in 0 until m || y !in 0 until n) return
    if (visited[x][y]) return
    if (grid[x][y] == 1) return

    visited[x][y] = true
    grid[x][y] = invalid
    for (d in directions) {
        val newX = x + d[0]
        val newY = y + d[1]
        dfs(grid, newX, newY, visited)
    }
}
```

* **Time Complexity**: `O(m * n)`.
* **Space Complexity**: `O(m * n)`.
