# [2241. Design An ATM Machine](https://leetcode.com/problems/design-an-atm-machine/description/)

## Greedy
When withdrawing, we should always try to use the largest denomination first. We can use the second largest denomination only if we run out of the largest one.

```kotlin
class ATM() {

    private val numbers = listOf(20, 50, 100, 200, 500)
    private val counts = LongArray(numbers.size)

    fun deposit(banknotesCount: IntArray) {
        for (i in banknotesCount.indices) {
            counts[i] += banknotesCount[i].toLong()
        }
    }

    /**
     * A, B, C
     * 1  1  3
     * We have to run out of A, then we can use B.
     * We can use next only if we use all of current one.
     * We can't use C until we use all B
     *
     * A, B, C
     * 1  0  3, we can use A then C
     */
    fun withdraw(amount: Int): IntArray {
        var amount = amount.toLong()
        val results = IntArray(numbers.size) // Temporary count array to withdraw, to ensure atomicity of the transaction.
        for (i in numbers.lastIndex downTo 0) {
            val requiredCount = (amount / numbers[i])
            val finalCount = minOf(requiredCount, counts[i])
            counts[i] -= finalCount
            results[i] = finalCount.toInt()
            amount -= 1L * finalCount * numbers[i]

            if (amount == 0L) break
        }
        return if (amount > 0L) { // If we can't withdraw, we rollback the counts.
            for (i in results.indices) {
                counts[i] += results[i]
            }
            intArrayOf(-1) 
        } else {
            results
        }
    }
}
```

### Pitfalls

```kotlin
fun withdraw(amount: Int): IntArray {
    var amount = amount.toLong()
    val results = IntArray(numbers.size) // Bug 3: Integer overflow
    for (i in numbers.lastIndex downTo 0) {
        if (counts[i] == 0) continue
        val count = (amount / numbers[i]).toInt()
        // Bug 1: 直接算出需要幾張鈔票，然後就扣除了，完全沒有檢查庫存是否有足夠那麼多張。
        counts[i] -= count
        results[i] = count
        amount -= 1L * count * numbers[i]

        if (amount == 0L) break
    }
    // Bug 2: 當我們發現無法湊齊時 (-1) 時，應該要把之前扣掉的錢加回來
    // 交易要滿足「原子性」：要嘛全部成功、要嘛全部失敗
    // 我們應該先用一個暫存的狀態去處理，確認可以湊齊時才更新最終狀態。
    return if (amount > 0L) intArrayOf(-1) else results
}
```
